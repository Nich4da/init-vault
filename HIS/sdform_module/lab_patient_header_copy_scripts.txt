3) COMPONENTS > EVENT SETTING > onCreated
============================================================

const s = this.vueState
const field = this

s.tran = null

s.patient = () => {
  return (s.tran && s.tran.vid && s.tran.vid.pid) || {}
}

s.visit = () => {
  return (s.tran && s.tran.vid) || {}
}

s.patientName = () => {
  const p = s.patient()
  const prefix = (p.prename && p.prename.label) || ''

  return (
    prefix +
    (p.p_fname || '') +
    ' ' +
    (p.p_lname || '')
  ).trim() || '-'
}

s.picUrl = () => {
  const pictures = s.patient().p_pic

  return (
    Array.isArray(pictures) &&
    pictures[0] &&
    pictures[0].url
  ) || ''
}

s.genderLabel = () => {
  const genderMap = {
    '1': 'ชาย',
    '2': 'หญิง'
  }

  return genderMap[String(s.patient().p_gender || '')] || '-'
}

s.aboLabel = () => {
  const bloodGroupMap = {
    '1': 'A',
    '2': 'B',
    '3': 'AB',
    '4': 'O',
    '5': 'ไม่ทราบ',
    '9': 'ไม่ทราบ'
  }

  const code = String(s.patient().p_abogroup || '')

  return bloodGroupMap[code] || code || '-'
}

s.visitDate = () => {
  return (
    s.visit().visit_date ||
    (s.tran && s.tran.checkin_at) ||
    '-'
  )
}

s.patientTypes = () => {
  const values = s.tran && s.tran.pttype
  return Array.isArray(values) ? values : []
}

s.allergies = () => {
  const values = s.tran && s.tran.allergy_tags
  return Array.isArray(values) ? values : []
}

s.setTran = (row) => {
  s.tran = row || null

  const form = field.getFormRef()

  if (form) {
    form.$labTran = s.tran
  }
}

s.clearTran = () => {
  s.setTran(null)
}

7) LIST VIEW > BUTTONS ROW > ปุ่มเรียก/เลือกผู้ป่วย > onClick
============================================================

หมายเหตุ:
- ใน Buttons Row ของ List View ข้อมูลแถวปัจจุบันใช้ตัวแปร dataRow
- ต้องวาง Component และตั้งชื่อ lab_patient_header ก่อน

const form = this.getFormRef()
const selectedTran = dataRow

form.$labTran = selectedTran

const patientHeader = form.getFieldRef('lab_patient_header')

if (
  patientHeader &&
  patientHeader.vueState &&
  patientHeader.vueState.setTran
) {
  patientHeader.vueState.setTran(selectedTran)
} else {
  this.alert(
    'ไม่พบ Component lab_patient_header',
    'warning'
  )
}
